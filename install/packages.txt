# Shell / Umbriel · paquetes para Debian 13 (trixie), mismo hardware
#
# La vía recomendada es ./install.sh, que aplica esta lista automáticamente.
# Este archivo queda como referencia manual y para revisar qué instala.
#
# Los paquetes de red, punto de acceso, VPN y Bluetooth los instala
# scripts/network.sh (no install-packages.sh) para no duplicarlos.
#
# Punto de partida recomendado: instalar Debian 13 con el entorno KDE Plasma
# (task-kde-desktop). Eso aporta Qt 6, KDE, Breeze, PipeWire,
# NetworkManager, BlueZ, UPower, Polkit, kdialog y el portal de KDE. SDDM se
# queda como respaldo del iniciador principal, Noctalia Greeter.
# Si se parte de una instalación mínima, instalar también los paquetes
# marcados como «base KDE» en la sección 3.

# ---------------------------------------------------------------------------
# 1. Repositorios
# ---------------------------------------------------------------------------
# Debian 13 + trixie-backports:
#   deb http://deb.debian.org/debian trixie-backports main
#
# Noctalia (Umbriel y su portal): seguir las instrucciones oficiales:
#   wget https://pkg.noctalia.dev/deb/nickh-archive-keyring.deb
#   sudo dpkg -i nickh-archive-keyring.deb
#   sudo wget -O /etc/apt/sources.list.d/noctalia-trixie.sources \
#     https://pkg.noctalia.dev/deb/noctalia-trixie.sources
#   sudo apt update
#
# Los archivos de ejemplo de este repositorio están en install/apt/.

# ---------------------------------------------------------------------------
# 2. Shell, compositor y portal
# ---------------------------------------------------------------------------
sudo apt install -t trixie-backports quickshell hyprlock
sudo apt install -t trixie-backports umbriel xdg-desktop-portal-umbriel

# Quickshell arrastra at-spi2-core, libatk1.0-0t64 y libatspi2.0-0t64 desde
# backports. Los GIR de ATK y AT-SPI deben venir del mismo origen: los de
# trixie exigen la versión exacta de trixie, así que instalar después una app
# GTK3 (por ejemplo un .deb como LocalSend) falla con «held broken packages».
# El instalador lo hace solo si libatk1.0-0t64 es de backports.
sudo apt install -t trixie-backports gir1.2-atk-1.0 gir1.2-atspi-2.0

# Noctalia Greeter es el iniciador de sesión por defecto. --no-install-recommends
# evita que fonts-noctalia-tabler instale el paquete `noctalia` (shell completo);
# este escritorio usa Umbriel.
sudo apt install --no-install-recommends -t trixie-backports \
  greetd noctalia-greeter

# El instalador lo configura con:
#   ./scripts/install-greeter.sh
# SDDM se queda instalado como respaldo y se puede recuperar con
#   sudo systemctl disable greetd && sudo systemctl enable sddm

# ---------------------------------------------------------------------------
# 3. Dependencias del shell que no siempre trae el entorno KDE
#    (la mayoría ya vienen con task-kde-desktop)
# ---------------------------------------------------------------------------
sudo apt install \
  ddcutil brightnessctl qrencode libnotify-bin fonts-inter-variable \
  wvkbd swayidle foot grim slurp \
  fonts-noto-cjk python3 curl \
  network-manager-openvpn wpasupplicant rfkill dnsmasq-base \
  power-profiles-daemon iputils-ping wl-clipboard \
  cliphist udiskie udisks2 ntfs-3g exfatprogs usbutils \
  qml6-module-qtquick-effects \
  qml6-module-qt-labs-folderlistmodel \
  xdg-utils desktop-file-utils \
  xdg-desktop-portal xdg-desktop-portal-gtk xdg-desktop-portal-kde \
  xwayland

# grim y slurp no son solo para pruebas: Print captura la pantalla y
# Shift+Print una región. fonts-noto-cjk cubre las letras japonesas, chinas y
# coreanas del widget de música; Inter no trae esos glifos. brightnessctl es
# el brillo del panel interno cuando ddcutil no sirve (portátiles).

# xdg-utils y desktop-file-utils son los que usan los postinst de los .deb para
# publicar su lanzador. Además, xdg-desktop-menu exige que existan
# /etc/xdg/menus y /usr/share/desktop-directories; install-packages.sh los crea
# antes de instalar paquetes. Sin ellos, un .deb como Spotify deja el binario y
# los iconos instalados pero no aparece en el lanzador («No writable system menu
# directory found»).

# wpasupplicant es imprescindible para que NetworkManager pueda escanear y
# conectar Wi-Fi; network-manager solo lo recomienda. rfkill permite comprobar
# si la radio está bloqueada. dnsmasq-base es el DHCP/DNS que NetworkManager
# usa para el punto de acceso compartido. Si Debian dejó la Ethernet en
# /etc/network/interfaces, NetworkManager la verá «unmanaged»:
#   sudo ./scripts/network.sh
#
# Audio Bluetooth de PipeWire: sin libspa-0.2-bluetooth los auriculares se
# emparejan pero no aparecen como salida ni micrófono, y la conexión puede
# fallar con br-connection-profile-unavailable. El plugin debe coincidir con
# la versión instalada de libspa-0.2-modules:
#   sudo apt install -t trixie-backports libspa-0.2-bluetooth

# PipeWire y Mesa deben venir de backports y en bloque. Mezclar versiones de
# trixie y backports rompe las dependencias exactas de libspa, pixman, Mesa y
# Wayland. El instalador ya lo hace así; esta es la versión manual:
#   sudo apt install -t trixie-backports \
#     pipewire pipewire-bin pipewire-pulse wireplumber \
#     libpipewire-0.3-0t64 libpipewire-0.3-modules libspa-0.2-modules
#   sudo apt install -t trixie-backports \
#     libgbm1 mesa-libgallium libgl1-mesa-dri libglx-mesa0 libegl-mesa0 \
#     mesa-va-drivers mesa-vulkan-drivers

# Base KDE ligera que usa el shell/Umbriel si no se instaló task-kde-desktop.
# Sin plasma-workspace: no hace falta KWin ni el escritorio Plasma.
# SDDM se queda como respaldo de Noctalia Greeter; sddm-theme-maya solo se
# instala si no hay ya un tema Breeze de SDDM.
#   sudo apt install --no-install-recommends \
#     sddm sddm-theme-maya libpam-systemd plasma-integration \
#     libkf6config-bin kdialog \
#     breeze breeze-icon-theme breeze-gtk-theme breeze-cursor-theme \
#     qt6-wayland qt6-qpa-plugins qt6-svg-plugins \
#     network-manager wpasupplicant upower polkitd bluez bluez-obexd \
#     libglib2.0-bin librsvg2-common
#
# El acento personalizado en apps KDE requiere plasma-apply-colorscheme
# (paquete plasma-workspace). Es opcional; sin él, las apps usan Breeze.

# ---------------------------------------------------------------------------
# 4. Dependencias para compilar el edit de Umbriel
# ---------------------------------------------------------------------------
sudo apt install -t trixie-backports \
  build-essential cmake git meson ninja-build pkg-config \
  libwlroots-0.20-dev libwayland-dev wayland-protocols \
  libdrm-dev libinput-dev libxkbcommon-dev \

  libeis-dev libseat-dev nlohmann-json3-dev

# ---------------------------------------------------------------------------
# 5. Firmware y gráficos de este equipo
# ---------------------------------------------------------------------------
# Paquetes de referencia para un equipo Intel + AMD. Ajustar la lista al
# hardware real: en un portátil cambian el firmware Wi-Fi y el driver de audio,
# y con NVIDIA hacen falta los paquetes de esa GPU. El instalador solo ejecuta
# esta sección con --with-firmware-author; es una lista del equipo del autor,
# no una detección automática.
sudo apt install \
  intel-microcode \
  firmware-amd-graphics firmware-iwlwifi firmware-realtek \
  firmware-sof-signed \
  mesa-utils intel-media-va-driver

# ---------------------------------------------------------------------------
# 6. Steam (librerías de 32 bits)
# ---------------------------------------------------------------------------
# Steam es un cliente de 32 bits. Sin libc6:i386 no arranca y muestra
# «you are missing the following 32-bit libraries: libc.so.6».
# Se instalan con -t trixie-backports para que las versiones i386 coincidan
# con el Mesa/Wayland de 64 bits de backports; si no, apt falla con
# «held broken packages». El instalador lo hace con scripts/install-steam-libs.sh.
sudo dpkg --add-architecture i386
sudo apt update
sudo apt install -t trixie-backports \
  libc6:i386 libegl1:i386 libgbm1:i386 libgl1:i386 libgl1-mesa-dri:i386 \
  steam-libs:amd64 steam-libs:i386

# Si usas el repositorio oficial de Steam, los metapaquetes se llaman
# steam-libs-amd64:amd64 y steam-libs-i386:i386; son equivalentes.
# Con GPU NVIDIA hacen falta además nvidia-driver-libs:i386 y su versión amd64.
# Esto instala solo las librerías; Steam (steam-launcher) se instala aparte.
